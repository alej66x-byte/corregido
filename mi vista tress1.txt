(function ($) {
    "use strict";

    $(function () {
        var $app = $("#tressApp");

        if (!$app.length) {
            return;
        }

        var $error = $("#tressError");
        var $exito = $("#tressExito");
        var $grupo = $("#tressGrupoSeleccionado");
        var $grupoInfo = $("#tressGrupoInfo");
        var $total = $("#tressTotal");
        var $grupoTotal = $("#tressGrupoTotal");

        var idGrupoActual = null;
        var cargando = false;
        var guardando = false;
        var detalleCargado = false;
        var cargandoDetalle = false;
        var modalPreparacion = null;
        var modalGrupo = null;

        function token() {
            return $app.find('input[name="__RequestVerificationToken"]')
                .first().val();
        }

        function mostrarError(texto) {
            $error.text(texto).prop("hidden", false);
        }

        function mostrarExito(texto) {
            $exito.text(texto).prop("hidden", false);
        }

        function limpiarMensajes() {
            $error.text("").prop("hidden", true);
            $exito.text("").prop("hidden", true);
        }

        function mensajeError(xhr, estado, defecto) {
            if (xhr.responseJSON && xhr.responseJSON.message) {
                return xhr.responseJSON.message;
            }

            if (estado === "timeout") {
                return "La solicitud tardó demasiado.";
            }

            return defecto;
        }

        function insignia(texto, clase) {
            return '<span class="tress-badge ' + clase + '">' +
                texto + "</span>";
        }

        var tablaContratados = $("#tblTressContratados").DataTable({
            data: [],
            autoWidth: false,
            pageLength: 10,
            dom: '<"tress-table-viewport"t>ip',
            order: [[3, "desc"], [0, "desc"]],

            columns: [
                { data: "Id_Vacante_Ext" },
                {
                    data: "NombreCompleto",
                    defaultContent: ""
                },
                {
                    data: "Puesto_Aplicado",
                    defaultContent: ""
                },
                {
                    data: "Fecha_Registro_Fmt",
                    defaultContent: ""
                },
                {
                    data: "Tress",
                    render: function (data, type, row) {
                        var curp = String(row.CURP || "").trim();
                        var existe = Number(data) === 1;
                        var texto = !curp
                            ? "Sin CURP"
                            : existe
                                ? "Existe en TRESS"
                                : "No encontrado";

                        if (type !== "display") {
                            return texto;
                        }

                        return insignia(
                            texto,
                            existe && curp
                                ? "tress-badge-green"
                                : "tress-badge-gray"
                        );
                    }
                },
                {
                    data: "EnGrupoExportacion",
                    render: function (data, type) {
                        var dentro = Number(data) === 1;
                        var texto = dentro ? "En el grupo" : "Sin agregar";

                        if (type !== "display") {
                            return texto;
                        }

                        return insignia(
                            texto,
                            dentro
                                ? "tress-badge-blue"
                                : "tress-badge-gray"
                        );
                    }
                },
                {
                    data: null,
                    orderable: false,
                    searchable: false,
                    render: function (data, type, row) {
                        if (type !== "display") {
                            return "";
                        }

                        var dentro = Number(row.EnGrupoExportacion) === 1;

                        return '<button type="button" ' +
                            'class="tress-row-action js-preparar">' +
                            (dentro ? "Editar" : "Agregar") +
                            "</button>";
                    }
                }
            ],

            language: {
                emptyTable: "No hay contratados para mostrar.",
                zeroRecords: "No se encontraron coincidencias.",
                info: "Mostrando _START_ a _END_ de _TOTAL_ contratados",
                infoEmpty: "Sin registros",
                infoFiltered: "(de _MAX_ contratados)",
                paginate: {
                    first: "Primera",
                    last: "Última",
                    next: "Siguiente",
                    previous: "Anterior"
                }
            }
        });

        var tablaGrupo = $("#tblTressGrupo").DataTable({
            data: [],
            autoWidth: false,
            pageLength: 8,
            dom: "tip",
            columns: [
                { data: "Id_Vacante_Ext" },
                {
                    data: "NombreCompleto",
                    defaultContent: ""
                },
                {
                    data: "NumeroEmpleado",
                    defaultContent: ""
                },
                {
                    data: "CodigoTurno",
                    defaultContent: ""
                },
                {
                    data: null,
                    orderable: false,
                    searchable: false,
                    render: function (data, type) {
                        if (type !== "display") {
                            return "";
                        }

                        return '<button type="button" ' +
                            'class="tress-row-action js-editar-grupo">' +
                            "Editar</button> " +
                            '<button type="button" ' +
                            'class="tress-row-action tress-row-action-danger ' +
                            'js-quitar-grupo">' +
                            "Quitar</button>";
                    }
                }
            ],

            language: {
                emptyTable: "Este grupo todavía no tiene candidatos.",
                zeroRecords: "No se encontraron coincidencias.",
                info: "_TOTAL_ candidatos",
                infoEmpty: "0 candidatos",
                paginate: {
                    next: "Siguiente",
                    previous: "Anterior"
                }
            }
        });

        function cargarGrupos() {
            return $.ajax({
                url: $app.data("grupos-url"),
                type: "POST",
                dataType: "json",
                data: {
                    __RequestVerificationToken: token()
                }
            })
                .done(function (grupos) {
                    $grupo.empty();

                    if (!Array.isArray(grupos) || !grupos.length) {
                        $grupo.append(
                            $("<option>", {
                                value: "",
                                text: "No hay grupos activos"
                            })
                        );

                        idGrupoActual = null;
                        limpiarTablas();
                        return;
                    }

                    $.each(grupos, function (_, item) {
                        $grupo.append(
                            $("<option>", {
                                value: item.IdGrupo,
                                text: item.Nombre +
                                    " (" + item.TotalCandidatos + ")"
                            })
                        );
                    });

                    idGrupoActual = Number(grupos[0].IdGrupo);
                    $grupo.val(idGrupoActual);
                    mostrarInfoGrupo(grupos[0]);

                    cargarTodo();
                })
                .fail(function (xhr, estado) {
                    mostrarError(mensajeError(
                        xhr,
                        estado,
                        "No se pudieron cargar los grupos."
                    ));
                });
        }

        function mostrarInfoGrupo(grupo) {
            if (!grupo) {
                $grupoInfo.text("Selecciona un grupo para comenzar.");
                return;
            }

            $grupoInfo.text(
                "Creado: " +
                (grupo.FechaCreacion || "") +
                " · Candidatos: " +
                grupo.TotalCandidatos
            );
        }

        function limpiarTablas() {
            tablaContratados.clear().draw();
            tablaGrupo.clear().draw();
            $total.text("Sin grupo");
            $grupoTotal.text("0 candidatos");
        }

        function cargarContratados() {
            if (!idGrupoActual) {
                limpiarTablas();
                return;
            }

            var busqueda = String($("#tressBusqueda").val() || "").trim();

            $.ajax({
                url: $app.data("consultar-url"),
                type: "POST",
                dataType: "json",
                timeout: 90000,
                data: {
                    idGrupo: idGrupoActual,
                    busqueda: busqueda,
                    __RequestVerificationToken: token()
                }
            })
                .done(function (respuesta) {
                    if (!Array.isArray(respuesta)) {
                        mostrarError(
                            "El servidor no devolvió una lista válida."
                        );
                        return;
                    }

                    tablaContratados.clear()
                        .rows.add(respuesta)
                        .draw();

                    $total.text(
                        respuesta.length +
                        (busqueda
                            ? " resultados"
                            : " solicitudes recientes")
                    );
                })
                .fail(function (xhr, estado) {
                    mostrarError(mensajeError(
                        xhr,
                        estado,
                        "No se pudieron consultar los contratados."
                    ));
                });
        }

        function cargarGrupo() {
            if (!idGrupoActual) {
                return;
            }

            $.ajax({
                url: $app.data("grupo-url"),
                type: "POST",
                dataType: "json",
                data: {
                    idGrupo: idGrupoActual,
                    __RequestVerificationToken: token()
                }
            })
                .done(function (respuesta) {
                    if (!Array.isArray(respuesta)) {
                        mostrarError(
                            "El servidor no devolvió el grupo correctamente."
                        );
                        return;
                    }

                    tablaGrupo.clear()
                        .rows.add(respuesta)
                        .draw();

                    $grupoTotal.text(
                        respuesta.length +
                        (respuesta.length === 1
                            ? " candidato"
                            : " candidatos")
                    );
                })
                .fail(function (xhr, estado) {
                    mostrarError(mensajeError(
                        xhr,
                        estado,
                        "No se pudo consultar el grupo."
                    ));
                });
        }

        function cargarTodo() {
            limpiarMensajes();
            cargarContratados();
            cargarGrupo();
        }

        function abrirModalPreparacion(idVacanteExt) {
            if (!idGrupoActual || cargandoDetalle || guardando) {
                return;
            }

            var elemento = document.getElementById("tressModal");

            modalPreparacion =
                bootstrap.Modal.getOrCreateInstance(elemento);

            $("#tressFormularioPreparacion")[0].reset();
            $("#tressModalError").text("").prop("hidden", true);
            $("#tressModalCargando").prop("hidden", false);
            $("#tressFormularioPreparacion input, " +
                "#tressFormularioPreparacion select")
                .prop("disabled", true);

            $("#btnGuardarPreparacionTress")
                .prop("disabled", true)
                .text("Guardar");

            modalPreparacion.show();
            cargandoDetalle = true;

            $.ajax({
                url: $app.data("detalle-url"),
                type: "POST",
                dataType: "json",
                data: {
                    idGrupo: idGrupoActual,
                    idVacanteExt: idVacanteExt,
                    __RequestVerificationToken: token()
                }
            })
                .done(function (respuesta) {
                    if (!respuesta.success || !respuesta.data) {
                        $("#tressModalError")
                            .text("No se pudo cargar el candidato.")
                            .prop("hidden", false);
                        return;
                    }

                    llenarModal(respuesta.data);
                })
                .fail(function (xhr, estado) {
                    $("#tressModalError")
                        .text(mensajeError(
                            xhr,
                            estado,
                            "No se pudo cargar el detalle."
                        ))
                        .prop("hidden", false);
                })
                .always(function () {
                    cargandoDetalle = false;
                    $("#tressModalCargando").prop("hidden", true);
                });
        }

        function llenarModal(data) {
            var form = $("#tressFormularioPreparacion")[0];

            Object.keys(data).forEach(function (nombre) {
                var elemento = form.elements.namedItem(nombre);

                if (elemento && data[nombre] !== null &&
                    data[nombre] !== undefined) {
                    elemento.value = data[nombre];
                }
            });

            $("#tressCandidatoNombre")
                .text(data.NombreCompleto || "Sin nombre");
            $("#tressCandidatoId")
                .text(data.Id_Vacante_Ext);
            $("#tressCandidatoCurp")
                .text(data.CURP || "Sin CURP");
            $("#tressCandidatoSexo")
                .text(data.Sexo || "Sin capturar");
            $("#tressCandidatoPuesto")
                .text(data.Puesto_Aplicado || "Sin puesto");

            var enGrupo =
                data.EnGrupo === true || Number(data.EnGrupo) === 1;

            $("#tressModalTitulo").text(
                enGrupo ? "Editar preparación" : "Agregar al grupo"
            );

            $("#btnGuardarPreparacionTress")
                .text(enGrupo ? "Guardar cambios" : "Guardar y agregar")
                .prop("disabled", false);

            $("#tressFormularioPreparacion input, " +
                "#tressFormularioPreparacion select")
                .prop("disabled", false);

            detalleCargado = true;
        }

        function guardarPreparacion(event) {
            event.preventDefault();

            if (!detalleCargado || guardando) {
                return;
            }

            var form = $("#tressFormularioPreparacion")[0];

            if (!form.reportValidity()) {
                return;
            }

            var datos = {};

            $("#tressFormularioPreparacion")
                .serializeArray()
                .forEach(function (item) {
                    datos[item.name] = item.value;
                });

            var salario = String(datos.SalarioDiario || "")
                .trim()
                .replace(",", ".");

            if (!/^\d{1,11}(\.\d{1,4})?$/.test(salario)) {
                $("#tressModalError")
                    .text("Captura un salario diario válido.")
                    .prop("hidden", false);
                return;
            }

            datos.SalarioDiario = salario;
            datos.__RequestVerificationToken = token();

            guardando = true;

            $("#btnGuardarPreparacionTress")
                .prop("disabled", true)
                .text("Guardando...");

            $.ajax({
                url: $app.data("guardar-url"),
                type: "POST",
                dataType: "json",
                data: datos
            })
                .done(function (respuesta) {
                    if (!respuesta.success) {
                        mostrarError(respuesta.message);
                        return;
                    }

                    bootstrap.Modal
                        .getInstance(document.getElementById("tressModal"))
                        .hide();

                    mostrarExito(
                        respuesta.message ||
                        "La preparación se guardó correctamente."
                    );

                    cargarTodo();
                })
                .fail(function (xhr) {
                    if (xhr.responseJSON &&
                        xhr.responseJSON.code === "EXPORTACION_PREVIA") {
                        Swal.fire({
                            icon: "warning",
                            title: "Candidato exportado anteriormente",
                            text: xhr.responseJSON.message,
                            showCancelButton: true,
                            confirmButtonText: "Agregar de todos modos",
                            cancelButtonText: "Cancelar"
                        }).then(function (resultado) {
                            if (resultado.isConfirmed) {
                                datos.confirmarExportado = true;
                                $("#tressFormularioPreparacion")
                                    .trigger("submit");
                            }
                        });

                        return;
                    }

                    $("#tressModalError")
                        .text(mensajeError(
                            xhr,
                            "",
                            "No se pudo guardar la preparación."
                        ))
                        .prop("hidden", false);
                })
                .always(function () {
                    guardando = false;

                    $("#btnGuardarPreparacionTress")
                        .prop("disabled", false)
                        .text("Guardar");
                });
        }

        function quitarDelGrupo(item) {
            Swal.fire({
                icon: "question",
                title: "¿Quitar del grupo?",
                text: "Los datos capturados se conservarán.",
                showCancelButton: true,
                confirmButtonText: "Sí, quitar",
                cancelButtonText: "Cancelar"
            }).then(function (resultado) {
                if (!resultado.isConfirmed) {
                    return;
                }

                $.ajax({
                    url: $app.data("quitar-url"),
                    type: "POST",
                    dataType: "json",
                    data: {
                        idGrupo: idGrupoActual,
                        idPreparacion: item.IdPreparacion,
                        versionFila: item.VersionFila,
                        __RequestVerificationToken: token()
                    }
                })
                    .done(function (respuesta) {
                        if (!respuesta.success) {
                            mostrarError(respuesta.message);
                            return;
                        }

                        mostrarExito(respuesta.message);
                        cargarTodo();
                    })
                    .fail(function (xhr, estado) {
                        mostrarError(mensajeError(
                            xhr,
                            estado,
                            "No se pudo retirar el candidato."
                        ));
                    });
            });
        }

        $("#tressGrupoSeleccionado").on("change", function () {
            idGrupoActual = Number(this.value) || null;

            var texto = $(this).find("option:selected").text();
            $grupoInfo.text("Grupo seleccionado: " + texto);

            cargarTodo();
        });

        $("#tressFormularioBusqueda").on("submit", function (event) {
            event.preventDefault();
            cargarContratados();
        });

        $("#btnActualizarTress").on("click", cargarTodo);

        $("#tblTressContratados tbody").on(
            "click",
            ".js-preparar",
            function () {
                var item = tablaContratados
                    .row($(this).closest("tr"))
                    .data();

                if (item) {
                    abrirModalPreparacion(item.Id_Vacante_Ext);
                }
            }
        );

        $("#tblTressGrupo tbody").on(
            "click",
            ".js-editar-grupo",
            function () {
                var item = tablaGrupo
                    .row($(this).closest("tr"))
                    .data();

                if (item) {
                    abrirModalPreparacion(item.Id_Vacante_Ext);
                }
            }
        );

        $("#tblTressGrupo tbody").on(
            "click",
            ".js-quitar-grupo",
            function () {
                var item = tablaGrupo
                    .row($(this).closest("tr"))
                    .data();

                if (item) {
                    quitarDelGrupo(item);
                }
            }
        );

        $("#tressFormularioPreparacion").on(
            "submit",
            guardarPreparacion
        );

        $("#btnNuevoGrupo").on("click", function () {
            $("#formNuevoGrupo")[0].reset();
            $("#nuevoGrupoError").prop("hidden", true);

            modalGrupo = bootstrap.Modal.getOrCreateInstance(
                document.getElementById("modalNuevoGrupo")
            );

            modalGrupo.show();
        });

        $("#formNuevoGrupo").on("submit", function (event) {
            event.preventDefault();

            var nombre = $("#nuevoGrupoNombre").val().trim();
            var observaciones =
                $("#nuevoGrupoObservaciones").val().trim();

            if (!nombre) {
                $("#nuevoGrupoError")
                    .text("Captura el nombre del grupo.")
                    .prop("hidden", false);
                return;
            }

            $("#btnGuardarGrupo")
                .prop("disabled", true)
                .text("Creando...");

            $.ajax({
                url: $app.data("crear-grupo-url"),
                type: "POST",
                dataType: "json",
                data: {
                    nombre: nombre,
                    observaciones: observaciones,
                    __RequestVerificationToken: token()
                }
            })
                .done(function (respuesta) {
                    if (!respuesta.success) {
                        $("#nuevoGrupoError")
                            .text(respuesta.message)
                            .prop("hidden", false);
                        return;
                    }

                    modalGrupo.hide();
                    mostrarExito(respuesta.message);
                    cargarGrupos();
                })
                .fail(function (xhr, estado) {
                    $("#nuevoGrupoError")
                        .text(mensajeError(
                            xhr,
                            estado,
                            "No se pudo crear el grupo."
                        ))
                        .prop("hidden", false);
                })
                .always(function () {
                    $("#btnGuardarGrupo")
                        .prop("disabled", false)
                        .text("Crear grupo");
                });
        });

        cargarGrupos();
    });
})(jQuery);
